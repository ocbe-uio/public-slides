1. Dataset & Model Specification
Dataset: Orthodont (built into nlme / R)

Dental growth distance (distance in mm) measured across age (8, 10, 12, 14 years) for 27 children clustered by Subject. 

library(lme4)

# Load built-in dataset (nlme package, pre-installed with R)
data(Orthodont, package = "nlme")

# Fit Linear Mixed Model (Random Intercept only)
fit_lmm <- lmer(distance ~ age + (1 | Subject), data = Orthodont)

# Alternative: Random Intercept + Random Slope for age
# fit_lmm <- lmer(distance ~ age + (age | Subject), data = Orthodont)

# Likelihood Ratio Test for random slops:
# m0 <- lmer(distance ~ age + (1 | Subject), data = Orthodont, REML = TRUE)
# m1 <- lmer(distance ~ age + (age | Subject), data = Orthodont, REML = TRUE)
# anova(m0, m1)
2. Printed Estimates
fixef() • confint() • summary()

Print fixed effects, standard errors, confidence intervals, and variance components.

# 1. Full model summary (coefficients, SEs, t-values)
summary(fit_lmm)

# 2. Fixed effects point estimates
fixef(fit_lmm)

# 3. 95% Confidence Intervals for fixed effects
confint(fit_lmm, parm = "beta_", method = "Wald")

# 4. Variance components (random effects & residual variance)
summary(fit_lmm)$varcor
confint(fit_lmm, parm = "theta_", method = "profile")
3. BLUPs (Best Linear Unbiased Predictors)
ranef() • coef()

ranef() extracts subject deviations (b_0j, b_1j); coef() extracts subject-specific totals (β + b_j). 

# 1. Subject random effect deviations from population mean (BLUPs)
ranef(fit_lmm)

# 2. Total subject-specific coefficients (Population Fixed + BLUP)
coef(fit_lmm)

# 3. Visualize BLUPs with prediction intervals
lattice::dotplot(ranef(fit_lmm, condVar = TRUE))
4. Model Checks & Diagnostics
Residuals • QQ Plots • Convergence

Verify homoscedasticity, normality of residuals, and normality of random effects.

# 1. Residuals vs. Fitted Plot (Homoscedasticity & Linearity)
plot(fit_lmm, type = c("p", "smooth"), main = "Residuals vs Fitted")

# 2. Residual Normality (Q-Q Plot)
qqnorm(residuals(fit_lmm), main = "Residuals Normal Q-Q")
qqline(residuals(fit_lmm), col = "red", lwd = 2)

# 3. Random Effects Normality (Q-Q Plots of BLUPs)
u <- ranef(fit_lmm)$Subject
qqnorm(u$`(Intercept)`, main = "BLUPs: Intercepts"); qqline(u$`(Intercept)`)

# 4. Convergence Check (0 = Successful convergence)
fit_lmm@optinfo$conv$opt
1. Dataset & Model Specification (nlme)
Dataset: Orthodont (built into nlme)

Dental growth distance (distance in mm) across age for 27 children clustered by Subject using nlme::lme(). Note: nlme separates fixed and random formulas. 

library(nlme)

# Load built-in dataset (pre-installed with base R)
data(Orthodont)

# Fit Linear Mixed Model (Random intercept only)
fit_nlme <- lme(fixed = distance ~ age, 
                random = ~ 1 | Subject, 
                data = Orthodont, 
                method = "REML")

# Alternative: Random Intercept + Random Slope for age
# fit_nlme <- lme(distance ~ age, random = ~ age | Subject, data = Orthodont

# Signature nlme extensions:
# Unequal variances across groups: weights = varIdent(form = ~ 1 | Sex)
# Autoregressive correlation:      correlation = corAR1(form = ~ 1 | Subject)
2. Printed Estimates
fixef() • intervals() • VarCorr()

Print fixed coefficients, confidence intervals, and variance-covariance structure.

# 1. Full model summary (fixed effects, t-table with approximate df)
summary(fit_nlme)

# 2. Fixed effects point estimates
fixef(fit_nlme)

# 3. 95% Confidence Intervals for fixed effects & variance components
intervals(fit_nlme)

# 4. Variance components (random effects SDs, correlation, residual SD)
VarCorr(fit_nlme)
3. BLUPs (Best Linear Unbiased Predictors)
ranef() • coef()

ranef() extracts subject deviations (b_0j, b_1j); coef() extracts subject-specific totals (β + b_j). 

# 1. Subject random deviations from population mean (BLUPs)
ranef(fit_nlme)

# 2. Total subject-specific coefficients (Population Fixed + BLUP)
coef(fit_nlme)

# 3. Visualize BLUPs with dotplot
plot(ranef(fit_nlme))
4. Model Checks & Diagnostics
Residuals • QQ Plots • nlme Features

Diagnostic plots

# 1. Standardized Residuals vs. Fitted Values
plot(fit_nlme, form = resid(., type = "p") ~ fitted(.), 
     abline = 0, main = "Standardized Residuals vs Fitted")

# 2. Residual Normality (Q-Q Plot)
qqnorm(fit_nlme, form = ~ resid(., type = "p"), main = "Residuals Normal Q-Q")

# 3. Random Effects Normality (Q-Q Plots of BLUPs)
qqnorm(fit_nlme, form = ~ ranef(.))
1. Dataset & Model Specification (Binary 0/1)
Dataset: ChickWeight (built into base R)

Binary outcome (heavy: whether weight > 100g, 0 = No, 1 = Yes) across Time (days) clustered by Chick. 

library(lme4)

# Load base R dataset and create binary 0/1 response
data(ChickWeight)
ChickWeight$heavy <- as.numeric(ChickWeight$weight > 100)

# Fit Logistic Mixed Model (Bernoulli outcome, Logit link)
fit_logit <- glmer(heavy ~ Time + (1 | Chick), 
                   data = ChickWeight, 
                   family = binomial(link = "logit"))
2. Printed Estimates & Odds Ratios
fixef() • exp() • Odds Ratios

Print log-odds estimates, exponentiated Odds Ratios (OR), and 95% confidence intervals.

# 1. Full model summary (z-values and p-values)
summary(fit_logit)

# 2. Log-odds estimates
fixef(fit_logit)

# 3. Odds Ratios (exp(beta))
exp(fixef(fit_logit))

# 4. 95% Confidence Intervals for Odds Ratios
exp(confint(fit_logit, parm = "beta_", method = "Wald"))
3. BLUPs (Subject-Specific Log-Odds Shifts)
ranef() • coef()

ranef() gives chick baseline deviation on the log-odds scale; coef() gives total individual intercept. 

# 1. Subject random deviations on log-odds scale (BLUPs: b_0j)
ranef(fit_logit)

# 2. Total subject-specific intercept on log-odds scale (beta_0 + b_0j)
coef(fit_logit)

# 3. Visualize BLUPs with prediction intervals
lattice::dotplot(ranef(fit_logit, condVar = TRUE))
4. Model Checks & Diagnostics
Pearson Residuals • Random Effect QQ

Inspect Pearson residuals across fitted probabilities and test BLUP normality.

# 1. Pearson Residuals vs. Fitted Probabilities
plot(fitted(fit_logit), residuals(fit_logit, type = "pearson"),
     xlab = "Fitted Probability", ylab = "Pearson Residuals",
     pch = 20, main = "Residuals vs Fitted")
abline(h = 0, lty = 2, col = "red")

# 2. Random Effects Normality (Q-Q Plot of Chick BLUPs)
qqnorm(ranef(fit_logit)$Chick[, 1], main = "BLUPs Q-Q Plot (Chick Intercepts)")
qqline(ranef(fit_logit)$Chick[, 1], col = "red", lwd = 2)

# 3. Convergence Check (0 = Successful convergence)
fit_logit@optinfo$conv$opt
1. Dataset & Model Specification (Counts)
Dataset: epil (built into MASS / R)

Repeated seizure counts (y) in a clinical trial across 59 epileptic patients (subject) comparing treatment (trt: placebo vs. progabide). 

library(lme4)

# Load built-in dataset (MASS package, pre-installed with R)
data(epil, package = "MASS")

# Fit Poisson Mixed Model (Count outcome, Log link)
fit_pois <- glmer(y ~ trt + (1 | subject), 
                  data = epil, 
                  family = poisson(link = "log"))

# For rate models with varying exposure/time, add an offset:
# glmer(y ~ trt + offset(log(time)) + (1 | subject), family = poisson)
2. Printed Estimates & Rate Ratios
fixef() • exp() • Rate Ratios

Fixed effects are on the log-rate scale. Exponentiate coefficients to obtain Rate Ratios (RR / IRR).

# 1. Full model summary (coefficients, z-values, p-values)
summary(fit_pois)

# 2. Log-rate coefficients (beta)
fixef(fit_pois)

# 3. Rate Ratios / Incidence Rate Ratios (exp(beta))
exp(fixef(fit_pois))

# 4. 95% Confidence Intervals for Rate Ratios
exp(confint(fit_pois, parm = "beta_", method = "Wald"))
3. BLUPs (Subject-Specific Log-Rate Shifts)
ranef() • coef()

ranef() yields subject random intercepts on the log-rate scale; coef() adds population baseline to individual subjects. 

# 1. Subject random deviations on log-rate scale (BLUPs: b_0j)
ranef(fit_pois)

# 2. Subject-specific total coefficients on log-rate scale (beta + b_0j)
coef(fit_pois)

# 3. Visualize subject BLUPs with prediction intervals
lattice::dotplot(ranef(fit_pois, condVar = TRUE))
4. Model Checks & Overdispersion Test
Overdispersion Check • Pearson Chi-Sq

Crucial for Poisson models (assumes Variance = Mean). A dispersion ratio > 1.4 suggests overdispersion. 

# 1. Overdispersion Check (Pearson Chi-Square / Residual DF)
pearson_res <- residuals(fit_pois, type = "pearson")
chisq_val   <- sum(pearson_res^2)
df_val      <- df.residual(fit_pois)
disp_ratio  <- chisq_val / df_val
p_value     <- pchisq(chisq_val, df = df_val, lower.tail = FALSE)

# Ratio > 1.4 with p < 0.05 indicates overdispersion
cat("Dispersion Ratio:", round(disp_ratio, 2), "| p-value:", round(p_value, 4), "\n")

# If overdispersed: add observation-level random effect (OLRE):
# epil$obs_id <- 1:nrow(epil)
# fit_olre <- glmer(y ~ trt + (1 | subject) + (1 | obs_id), data = epil, family = poisson)

# 2. Residuals vs. Fitted Plot
plot(fit_pois, type = c("p", "smooth"), 
     xlab = "Fitted Values", ylab = "Pearson Residuals",
     main = "Residuals vs Fitted (epil)")

# 3. Random Effect Normality (Q-Q Plot of BLUPs)
qqnorm(ranef(fit_pois)$subject[, 1], main = "BLUPs Q-Q Plot (Subject Effects)")
qqline(ranef(fit_pois)$subject[, 1], col = "red", lwd = 2)

# 4. Convergence Check (0 = Successful convergence)
fit_pois@optinfo$conv$opt

Convert Wide to Long Format

Mixed models in R (lme4) require data in long format, where each row represents one observation for a subject, with a subject ID column added.

Wide Format
id  visit_1  visit_2  visit_3
 1       12       15       19
 2       14       17       22
→
Long Format (for Mixed Models)
id  visit  score
 1      1     12
 1      2     15
 1      3     19
 2      1     14
 ...
R Code Example
# Sample Wide Data
df_wide <- data.frame(
  id      = 1:3,
  visit_1 = c(12, 14, 11),
  visit_2 = c(15, 17, 13),
  visit_3 = c(19, 22, 16)
)

# Option A: tidyr::pivot_longer (Recommended / Modern)
library(tidyr)
df_long <- pivot_longer(
  df_wide,
  cols            = starts_with("visit_"),  # columns to pivot
  names_to        = "visit",                # new column name
  names_prefix    = "visit_",               # drop "visit_" prefix
  names_transform = as.numeric,             # convert to numeric (1, 2, 3)
  values_to       = "score"                 # name of new value column
)

# Option B: Base R reshape (No packages required)
df_long_base <- reshape(
  df_wide,
  direction = "long",
  varying   = c("visit_1", "visit_2", "visit_3"),
  v.names   = "score",
  timevar   = "visit",
  times     = 1:3,                          # numeric 1, 2, 3 instead of strings
  idvar     = "id"
)

Plot Data Example (Orthodont Dataset)

Before fitting mixed models, visualize individual growth trajectories (spaghetti plot) and group mean trends across age to inform random intercepts, random slopes, and fixed effects.

What to look for:
• Vertical spread at baseline (Age 8):
Subject variation → random intercept (1 | Subject)
• Non-parallel lines:
Different individual growth rates → random slope (age | Subject)
R Plotting Code
# Load built-in dataset
data(Orthodont, package = "nlme")

# -------------------------------------------------------------
# 1. ggplot2: Spaghetti Plot with Group Mean Trajectories
# -------------------------------------------------------------
library(ggplot2)

ggplot(Orthodont, aes(x = age, y = distance, group = Subject, color = Sex)) +
  geom_line(alpha = 0.45, linewidth = 0.8) +
  geom_point(size = 1.6, alpha = 0.7) +
  # Bold group mean trajectory lines for Boys vs Girls:
  stat_summary(aes(group = Sex), fun = mean, geom = "line", linewidth = 1.5) +
  scale_color_manual(values = c("Male" = "#2563eb", "Female" = "#e11d48")) +
  labs(title = "Orthodont: Individual Trajectories & Sex Mean Trends",
       x = "Age (years)", y = "Distance (mm)", color = "Sex") +
  theme_minimal()

# -------------------------------------------------------------
# 2. lattice: Trellis Plot (Subject-by-Subject Panels)
# -------------------------------------------------------------
library(lattice)

# Individual panels per child (built-in with nlme / base R)
xyplot(distance ~ age | Subject, data = Orthodont,
       type = c("g", "p", "r"), # grid, points, linear fit
       layout = c(9, 3),        # 9 cols, 3 rows for 27 subjects
       xlab = "Age (years)", ylab = "Distance (mm)",
       main = "Individual Linear Growth Fits by Subject")
